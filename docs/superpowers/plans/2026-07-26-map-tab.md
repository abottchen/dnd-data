# Map Tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a "The Map" tab to the rendered site that displays the party's annotated player map of Chult in a deep pan-and-zoom viewer.

**Architecture:** A new `build/mapimage.py` re-encodes the gitignored 17 MB source under `data/` into a committed full-resolution JPEG in `site/images/`, running automatically inside `python -m build apply` right before the render and standalone via `python -m build map`. A new `validate_map` gates the render on that artifact existing. A new Jinja partial adds the panel; a new IIFE in the page's single inline script drives the viewer.

**Tech Stack:** Python 3 + Pillow + Jinja2, pytest, vanilla ES5-ish browser JS (no build step, no dependencies), hand-written CSS in `site/styles.css`.

## Global Constraints

- Source image: `data/chult-player-map.jpg`, 6445 × 8640, ~17 MB. Gitignored — never commit it.
- Served derivative: `site/images/chult-map.jpg`, full resolution, quality 80, progressive, `subsampling=0` (4:4:4), EXIF stripped. Committed.
- Zoom ceiling is derived from the decoded image: `max(6, (naturalWidth / clientWidth) * 2)` — roughly 14× on desktop. Never hardcode a flat cap.
- Scale 1 is fit-to-frame and is the minimum.
- Real player names must never reach the rendered page (see CLAUDE.md Privacy). Nothing in this feature touches player data.
- The site is desktop-first (`html, body { min-width: 1200px }`). Do not introduce responsive breakpoints; touch support is for tablets in landscape and trackpad gestures.
- All new colors and type come from the existing tokens in `site/styles.css` (`--ink-deep`, `--brass`, `--brass-hi`, `--brass-dim`, `--paper`, `--paper-dim`, `--rule`, `--rule-strong`).
- Run everything through the repo venv: `.venv/bin/python`, `.venv/bin/pytest`.
- Git hooks are active (`core.hooksPath=.githooks`). Do not use `--no-verify`.
- Commit messages: conventional commits, no trailers, no reference to Claude or any AI tool.

```python
"""mapimage.py — the web derivative of the party's annotated map of Chult.

The source under data/ is a ~17 MB, 55-megapixel export and is gitignored
like everything else in that directory; the committed artifact is a
re-encoded, full-resolution JPEG in site/images/. Full resolution is the
point — the viewer zooms past native pixel density so the handwritten
annotations stay legible.

Re-encoding 55 megapixels takes seconds, and most builds carry no new map,
so this is a no-op unless the source is newer than the derivative.
"""
from __future__ import annotations

from pathlib import Path

from .paths import REPO_ROOT, data_dir

SOURCE_NAME = "chult-player-map.jpg"
OUTPUT_NAME = "chult-map.jpg"

# 4:4:4 chroma (subsampling=0) costs a little size and buys crisp edges on the
# colored annotation strokes, which is the whole reason to keep full res.
QUALITY = 80
SUBSAMPLING = 0

STATUS_REBUILT = "rebuilt"
STATUS_SKIPPED = "skipped"
STATUS_NO_SOURCE = "no-source"


def source_path() -> Path:
    return data_dir() / SOURCE_NAME


def output_path() -> Path:
    return REPO_ROOT / "site" / "images" / OUTPUT_NAME


def fmt_bytes(n: int) -> str:
    if n < 1024:
        return f"{n} B"
    for unit in ("KB", "MB", "GB"):
        n /= 1024
        if n < 1024:
            return f"{n:.1f} {unit}"
    return f"{n:.1f} TB"


def _size_or_none(p: Path) -> int | None:
    return p.stat().st_size if p.exists() else None


def prepare_map(force: bool = False, *, src: Path | None = None,
                out: Path | None = None) -> dict:
    """Re-encode the map source into site/images/, unless it is already current.

    Returns {"status", "src_bytes", "out_bytes"}. A missing source is not an
    error: a clone without data/ still has the committed derivative, and a
    missing derivative is caught by validate_map at render time.
    """
    src = src or source_path()
    out = out or output_path()

    if not src.exists():
        return {"status": STATUS_NO_SOURCE, "src_bytes": None,
                "out_bytes": _size_or_none(out)}

    if not force and out.exists() and out.stat().st_mtime >= src.stat().st_mtime:
        return {"status": STATUS_SKIPPED, "src_bytes": src.stat().st_size,
                "out_bytes": out.stat().st_size}

    from PIL import Image

    with Image.open(src) as im:
        rgb = im.convert("RGB")   # loads the pixels; safe to use after close
    out.parent.mkdir(parents=True, exist_ok=True)
    rgb.save(out, format="JPEG", quality=QUALITY, optimize=True,
             progressive=True, subsampling=SUBSAMPLING)

    return {"status": STATUS_REBUILT, "src_bytes": src.stat().st_size,
            "out_bytes": out.stat().st_size}
```

def validate_map(images_dir: Path, filename: str = "chult-map.jpg") -> list[ValidationError]:
    """The player-map derivative must exist in site/images/ (see build/mapimage.py)."""
    if (images_dir / filename).exists():
        return []
    return [ValidationError(
        KIND_MISSING, "map", (filename,),
        field=f"'{filename}' not found in site/images/ — run "
              f"`python -m build map` to build it from data/chult-player-map.jpg",
    )]

def _cmd_map(args) -> int:
    from .mapimage import prepare_map, fmt_bytes, source_path, output_path

### Task 6: The viewer

**Files:**
- Modify: `build/templates/_script.html` (append a new IIFE at the end)

**Interfaces:**

- Produces: no exports — a self-contained IIFE

- [ ] **Step 1: Append the IIFE**

Add at the end of `build/templates/_script.html`:

```js
    (function () {
      // Map tab: deep pan/zoom over the company's annotated chart of Chult.
      // Scale 1 is fit-to-frame (the <img> is CSS-fitted by max-width/height);
      // the ceiling comes from the decoded image, so the viewer can push to
      // twice native pixel density instead of a hardcoded cap. The full
      // multi-megabyte JPEG is fetched only when the tab is first opened.
      const panel = document.getElementById('map');
      if (!panel) return;
      const frame = document.getElementById('map-frame');
      const viewport = document.getElementById('map-viewport');
      const img = document.getElementById('map-image');
      const loading = document.getElementById('map-loading');
      const gaugeFill = document.getElementById('map-gauge-fill');
      const gaugeNative = document.getElementById('map-gauge-native');
      const gaugeVal = document.getElementById('map-gauge-val');

      const MIN_SCALE = 1;
      let maxScale = 6;          // replaced by measure() once the image decodes
      let scale = 1, panX = 0, panY = 0;
      let pending = null;

      function clampPan() {
        const vw = viewport.clientWidth, vh = viewport.clientHeight;
        const iw = img.clientWidth * scale, ih = img.clientHeight * scale;
        if (iw <= vw) { panX = 0; } else { const m = (iw - vw) / 2; panX = Math.max(-m, Math.min(m, panX)); }
        if (ih <= vh) { panY = 0; } else { const m = (ih - vh) / 2; panY = Math.max(-m, Math.min(m, panY)); }
      }

      function paint() {
        pending = null;
        clampPan();
        img.style.transform = 'translate3d(' + panX + 'px,' + panY + 'px,0) scale(' + scale + ')';
        const span = Math.log(maxScale / MIN_SCALE);
        const pct = span > 0 ? (Math.log(scale / MIN_SCALE) / span) * 100 : 0;
        gaugeFill.style.width = Math.max(0, Math.min(100, pct)) + '%';
        gaugeVal.textContent = '×' + scale.toFixed(1);
      }

      // Wheel and pointermove arrive far faster than frames; collapse to one paint.
      function schedule() { if (pending === null) pending = requestAnimationFrame(paint); }

      function zoomTo(next, cx, cy) {
        next = Math.max(MIN_SCALE, Math.min(maxScale, next));
        if (cx !== undefined && cy !== undefined) {
          const r = next / scale;
          panX = cx - r * (cx - panX);
          panY = cy - r * (cy - panY);
        }
        scale = next;
        schedule();
      }

      function offsetFromCenter(clientX, clientY) {
        const r = viewport.getBoundingClientRect();
        return [clientX - r.left - r.width / 2, clientY - r.top - r.height / 2];
      }

      function reset() { scale = 1; panX = 0; panY = 0; schedule(); }

      function measure() {
        if (!img.naturalWidth || !img.clientWidth) return;
        // Native = the scale at which one image pixel covers one CSS pixel.
        const native = img.naturalWidth / img.clientWidth;
        maxScale = Math.max(6, native * 2);
        const span = Math.log(maxScale / MIN_SCALE);
        const at = span > 0 ? (Math.log(Math.max(native, MIN_SCALE) / MIN_SCALE) / span) * 100 : 100;
        gaugeNative.style.left = Math.min(100, at) + '%';
        if (scale > maxScale) scale = maxScale;
        schedule();
      }

      let requested = false;
      function ensureLoaded() {
        if (requested || !panel.classList.contains('active')) return;
        requested = true;
        img.addEventListener('load', function () { loading.remove(); measure(); }, { once: true });
        img.addEventListener('error', function () {
          loading.textContent = 'The chart could not be loaded.';
        }, { once: true });
        img.src = img.dataset.src;
      }
      // The tab switcher's hashchange listener is registered first, so the
      // panel already carries .active by the time this one runs.
      window.addEventListener('hashchange', ensureLoaded);
      ensureLoaded();

      viewport.addEventListener('wheel', function (e) {
        e.preventDefault();
        const c = offsetFromCenter(e.clientX, e.clientY);
        // ctrl+wheel is how a trackpad pinch reports — same gesture, finer steps.
        const step = e.ctrlKey ? Math.exp(-e.deltaY * 0.01) : (e.deltaY > 0 ? 0.85 : 1.18);
        zoomTo(scale * step, c[0], c[1]);
      }, { passive: false });

      const pointers = new Map();
      let startX = 0, startY = 0, startPanX = 0, startPanY = 0;
      let pinchDist = 0, pinchScale = 1;

      function seatPan(p) { startX = p.clientX; startY = p.clientY; startPanX = panX; startPanY = panY; }

      viewport.addEventListener('pointerdown', function (e) {
        if (e.pointerType === 'mouse' && e.button !== 0) return;
        viewport.setPointerCapture(e.pointerId);
        pointers.set(e.pointerId, e);
        if (pointers.size === 1) {
          seatPan(e);
        } else if (pointers.size === 2) {
          const p = [...pointers.values()];
          pinchDist = Math.hypot(p[0].clientX - p[1].clientX, p[0].clientY - p[1].clientY);
          pinchScale = scale;
        }
        e.preventDefault();
      });

      viewport.addEventListener('pointermove', function (e) {
        if (!pointers.has(e.pointerId)) return;
        pointers.set(e.pointerId, e);
        if (pointers.size === 1) {
          panX = startPanX + (e.clientX - startX);
          panY = startPanY + (e.clientY - startY);
          schedule();
        } else if (pointers.size === 2 && pinchDist > 0) {
          const p = [...pointers.values()];
          const dist = Math.hypot(p[0].clientX - p[1].clientX, p[0].clientY - p[1].clientY);
          const c = offsetFromCenter((p[0].clientX + p[1].clientX) / 2,
                                     (p[0].clientY + p[1].clientY) / 2);
          zoomTo(pinchScale * (dist / pinchDist), c[0], c[1]);
        }
      });

      function endPointer(e) {
        if (!pointers.delete(e.pointerId)) return;
        // Second finger lifted: re-seat the pan origin so the map doesn't jump.
        if (pointers.size === 1) seatPan([...pointers.values()][0]);
      }
      viewport.addEventListener('pointerup', endPointer);
      viewport.addEventListener('pointercancel', endPointer);

      viewport.addEventListener('dblclick', function (e) {
        e.preventDefault();
        const c = offsetFromCenter(e.clientX, e.clientY);
        zoomTo(scale * (e.shiftKey ? 0.5 : 2), c[0], c[1]);
      });

      viewport.addEventListener('keydown', function (e) {
        const step = e.shiftKey ? 200 : 60;
        if (e.key === 'ArrowLeft') panX += step;
        else if (e.key === 'ArrowRight') panX -= step;
        else if (e.key === 'ArrowUp') panY += step;
        else if (e.key === 'ArrowDown') panY -= step;
        else if (e.key === '+' || e.key === '=') zoomTo(scale * 1.3);
        else if (e.key === '-' || e.key === '_') zoomTo(scale * 0.75);
        else if (e.key === '0') reset();
        else return;
        e.preventDefault();
        schedule();
      });

      document.getElementById('map-zoom-in').addEventListener('click', function () { zoomTo(scale * 1.3); });
      document.getElementById('map-zoom-out').addEventListener('click', function () { zoomTo(scale * 0.75); });
      document.getElementById('map-zoom-reset').addEventListener('click', reset);
      document.getElementById('map-fullscreen').addEventListener('click', function () {
        if (document.fullscreenElement) document.exitFullscreen();
        else if (frame.requestFullscreen) frame.requestFullscreen();
      });

      // The fit basis changes with the frame, so the ceiling and the 1:1 tick
      // have to be recomputed whenever it resizes.
      window.addEventListener('resize', measure);
      document.addEventListener('fullscreenchange', measure);
    })();
```
